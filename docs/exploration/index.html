-- Rebuild the missing SPN-Aero candidate input from recovered source material.
--
-- Taxonomy: config/space-exploration8-taxonomy.yml
-- Source: Drive aerospace-patent-trends-draft.md, Appendix B, Table B-1
-- Grain: one row per DOCDB simple family x recovered technology field
-- Window: earliest priority date 2010-01-01 through 2020-12-31
--
-- Important corrections:
--   * The source contains eight clusters, not nine.
--   * Y10S903 is a current hybrid-electric-vehicle code and is excluded.
--   * Broad instrument/robot/life-saving classes require B64G co-classification.
--   * Lunar and planetary clusters require a matching title keyword.

CREATE OR REPLACE TABLE
  `aeropatent-research.aeropatent_research.candidates_snapshot`
CLUSTER BY tech_field, family_id
OPTIONS (
  description = 'Recovered space_exploration8_recovered_v1 candidate families; family_id x tech_field'
) AS
WITH publications AS (
  SELECT
    p.family_id,
    p.publication_number,
    p.priority_date,
    LOWER(ARRAY_TO_STRING(
      ARRAY(
        SELECT text
        FROM UNNEST(p.title_localized)
        WHERE text IS NOT NULL
      ),
      ' '
    )) AS title_text,
    ARRAY_CONCAT(
      ARRAY(
        SELECT AS STRUCT
          'CPC' AS system,
          REGEXP_REPLACE(UPPER(code), r'[^A-Z0-9/]', '') AS code
        FROM UNNEST(p.cpc)
        WHERE code IS NOT NULL
      ),
      ARRAY(
        SELECT AS STRUCT
          'IPC' AS system,
          REGEXP_REPLACE(UPPER(code), r'[^A-Z0-9/]', '') AS code
        FROM UNNEST(p.ipc)
        WHERE code IS NOT NULL
      )
    ) AS codes
  FROM `patents-public-data.patents.publications` AS p
  WHERE p.family_id IS NOT NULL
    AND p.priority_date BETWEEN 20100101 AND 20201231
),
features AS (
  SELECT
    *,
    EXISTS (
      SELECT 1 FROM UNNEST(codes)
      WHERE STARTS_WITH(code, 'B64G')
    ) AS has_space_anchor
  FROM publications
),
publication_matches AS (
  SELECT
    f.*,
    m.tech_field,
    m.tech_field_name,
    ARRAY(
      SELECT DISTINCT CONCAT(code.system, ':', code.code)
      FROM UNNEST(f.codes) AS code
      WHERE CASE m.tech_field
        WHEN 'DS-1' THEN
          STARTS_WITH(code.code, 'B64G1/10')
          OR STARTS_WITH(code.code, 'B64G1/22')
          OR STARTS_WITH(code.code, 'B64G1/58')
          OR STARTS_WITH(code.code, 'B64G1/62')
          OR STARTS_WITH(code.code, 'B64G4')
        WHEN 'DS-2' THEN
          STARTS_WITH(code.code, 'B64G1/1064')
          OR STARTS_WITH(code.code, 'B64G1/1071')
          OR STARTS_WITH(code.code, 'B64G1/22')
          OR STARTS_WITH(code.code, 'B64G1/64')
        WHEN 'DS-3' THEN
          STARTS_WITH(code.code, 'B64G1/40')
          OR STARTS_WITH(code.code, 'F02K9')
          OR STARTS_WITH(code.code, 'H05H1/54')
        WHEN 'DS-4' THEN
          STARTS_WITH(code.code, 'H04B10/118')
          OR STARTS_WITH(code.code, 'H04B7/185')
          OR STARTS_WITH(code.code, 'G01C21/24')
          OR STARTS_WITH(code.code, 'B64G1/1007')
          OR STARTS_WITH(code.code, 'B64G1/1014')
        WHEN 'DS-5' THEN
          STARTS_WITH(code.code, 'B64G1/58')
          OR STARTS_WITH(code.code, 'B64G1/62')
          OR STARTS_WITH(code.code, 'B64G1/14')
        WHEN 'DS-6' THEN
          STARTS_WITH(code.code, 'B64G1/46')
          OR STARTS_WITH(code.code, 'B64G1/48')
          OR STARTS_WITH(code.code, 'B64G1/50')
          OR STARTS_WITH(code.code, 'B64G1/54')
          OR STARTS_WITH(code.code, 'B64G1/60')
          OR (f.has_space_anchor AND STARTS_WITH(code.code, 'A62B'))
        WHEN 'DS-7' THEN
          (
            f.has_space_anchor
            AND (
              STARTS_WITH(code.code, 'G01N')
              OR STARTS_WITH(code.code, 'G01V')
              OR STARTS_WITH(code.code, 'G02B')
              OR STARTS_WITH(code.code, 'G01R33')
            )
          )
        WHEN 'DS-8' THEN
          STARTS_WITH(code.code, 'B64G4')
          OR (f.has_space_anchor AND STARTS_WITH(code.code, 'B25J9'))
        ELSE FALSE
      END
      ORDER BY 1
    ) AS matched_codes
  FROM features AS f
  CROSS JOIN UNNEST([
    STRUCT(
      'DS-1' AS tech_field,
      '달탐사' AS tech_field_name,
      REGEXP_CONTAINS(
        f.title_text,
        r'(lunar|moon|selen|달[[:space:]]*(탐사|궤도|착륙|표면)|월면|월궤도|월착륙)'
      )
      AND EXISTS (
        SELECT 1 FROM UNNEST(f.codes)
        WHERE STARTS_WITH(code, 'B64G1/10')
          OR STARTS_WITH(code, 'B64G1/22')
          OR STARTS_WITH(code, 'B64G1/58')
          OR STARTS_WITH(code, 'B64G1/62')
          OR STARTS_WITH(code, 'B64G4')
      ) AS is_match
    ),
    STRUCT(
      'DS-2',
      '행성탐사',
      REGEXP_CONTAINS(
        f.title_text,

-- Build the fixed 96-family pilot sample for the multisource bibliography study.
-- Source taxonomy: config/space-exploration8-taxonomy.yml
-- Unit: one DOCDB simple family. Multi-label technology fields are retained,
-- while the lexicographically first field is used only for stratification.

CREATE OR REPLACE TABLE
  `aeropatent-research.aeropatent_research.multisource_sample_v1`
CLUSTER BY primary_tech_field, priority_period, family_id
OPTIONS (
  description = 'Deterministic 96-family aerospace multisource bibliography pilot; 2026-09-10'
) AS
WITH candidate_families AS (
  SELECT
    family_id,
    ARRAY_AGG(DISTINCT tech_field ORDER BY tech_field) AS tech_fields,
    ARRAY_AGG(DISTINCT tech_field_name ORDER BY tech_field_name) AS tech_field_names,
    MIN(priority_date) AS priority_date,
    MIN(priority_year) AS priority_year
  FROM `aeropatent-research.aeropatent_research.candidates_snapshot`
  GROUP BY family_id
),
family_rollup AS (
  SELECT
    c.family_id,
    c.tech_fields,
    c.tech_field_names,
    c.tech_fields[SAFE_OFFSET(0)] AS primary_tech_field,
    c.priority_date,
    c.priority_year,
    CASE
      WHEN c.priority_year BETWEEN 2010 AND 2013 THEN 'P1'
      WHEN c.priority_year BETWEEN 2014 AND 2017 THEN 'P2'
      WHEN c.priority_year BETWEEN 2018 AND 2020 THEN 'P3'
    END AS priority_period,
    ARRAY_AGG(DISTINCT p.publication_number IGNORE NULLS ORDER BY p.publication_number) AS publication_numbers,
    ARRAY_AGG(DISTINCT p.application_number IGNORE NULLS ORDER BY p.application_number) AS application_numbers,
    ARRAY_CONCAT_AGG(ARRAY(SELECT value FROM UNNEST(p.assignee) AS value WHERE value IS NOT NULL)) AS applicant_names_raw,
    ARRAY_CONCAT_AGG(ARRAY(SELECT value.name FROM UNNEST(p.assignee_harmonized) AS value WHERE value.name IS NOT NULL)) AS applicant_names_harmonized,
    ARRAY_CONCAT_AGG(ARRAY(SELECT value.code FROM UNNEST(p.ipc) AS value WHERE value.code IS NOT NULL)) AS ipc_codes,
    ARRAY_CONCAT_AGG(ARRAY(SELECT value.code FROM UNNEST(p.cpc) AS value WHERE value.code IS NOT NULL)) AS cpc_codes,
    LOGICAL_OR(p.country_code IN ('EP', 'WO')) AS has_ep_or_wo_member,
    ARRAY_AGG(
      IF(
        p.country_code = 'KR',
        STRUCT(
          p.publication_number AS publication_number,
          p.application_number AS application_number,
          p.kind_code AS kind_code,
          p.publication_date AS publication_date,
          p.grant_date AS grant_date
        ),
        NULL
      )
      IGNORE NULLS
      ORDER BY
        CASE
          WHEN p.country_code = 'KR' AND p.kind_code LIKE 'A%' THEN 0
          WHEN p.country_code = 'KR' AND p.grant_date > 0 THEN 1
          ELSE 2
        END,
        IFNULL(p.publication_date, 99999999),
        p.publication_number
      LIMIT 1
    )[SAFE_OFFSET(0)] AS kr_anchor
  FROM candidate_families AS c
  JOIN `patents-public-data.patents.publications` AS p
    USING (family_id)
  GROUP BY
    c.family_id,
    c.tech_fields,
    c.tech_field_names,
    primary_tech_field,
    c.priority_date,
    c.priority_year,
    priority_period
),
eligible AS (
  SELECT *
  FROM family_rollup
  WHERE priority_period IS NOT NULL
    AND kr_anchor.publication_number IS NOT NULL
    AND kr_anchor.application_number IS NOT NULL
),
sampled AS (
  SELECT
    *,
    ROW_NUMBER() OVER (
      PARTITION BY primary_tech_field, priority_period
      ORDER BY SHA256(CONCAT('20260910|', family_id)), family_id
    ) AS stratum_order,
    COUNT(*) OVER (
      PARTITION BY primary_tech_field, priority_period
    ) AS stratum_population_n
  FROM eligible
)
SELECT
  family_id,
  tech_fields,
  tech_field_names,
  primary_tech_field,
  priority_date,
  priority_year,
  priority_period,
  publication_numbers,
  application_numbers,
  ARRAY(SELECT DISTINCT value FROM UNNEST(applicant_names_raw) AS value ORDER BY value) AS applicant_names_raw,
  ARRAY(SELECT DISTINCT value FROM UNNEST(applicant_names_harmonized) AS value ORDER BY value) AS applicant_names_harmonized,
  ARRAY(SELECT DISTINCT value FROM UNNEST(ipc_codes) AS value ORDER BY value) AS ipc_codes,
  ARRAY(SELECT DISTINCT value FROM UNNEST(cpc_codes) AS value ORDER BY value) AS cpc_codes,
  kr_anchor.publication_number AS kr_anchor_publication,
  kr_anchor.application_number AS kr_anchor_application,
  kr_anchor.kind_code AS kr_anchor_kind,
  stratum_population_n,
  stratum_order,
  DATE '2026-09-10' AS snapshot_date,
  'space_exploration8_recovered_v1' AS taxonomy_version,
  'sha256-20260910-v1' AS sampling_rule_version
FROM sampled
WHERE stratum_order <= 4;
